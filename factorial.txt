import code

import ollama
import re
def generate_function():
    prompt="""
    write a python function to generate factorial of a number named is_factorial(num).
    take the integer as an input.
    return the factorial of each number.

    return only python function.
    do not include any explanation or markdowns.
    """

    response=ollama.generate(
          model="qwen2.5-coder:0.5b",
          prompt=prompt
     )
    generated_code=response["response"].strip()

    match=re.match(
          r"```(?:python)?\s*(.*?)```",
          generated_code,
          re.DOTALL
     )

    if match:
          generated_code=match.group(1).strip()
    return generated_code

def test_function(code):
    namespace={}

    try:
        exec(code,namespace)
        is_factorial=namespace["is_factorial"]

        test_cases=[
               (0, 1),
               (1, 1),
               (2, 2),
               (3, 6),
               (4, 24)
          ]
        print("testing generated function")
        print("-"*40)
        passed=0

        for num, expected in test_cases:
            result=is_factorial(num)
            print(
                f"{num}={result}"
                f"(Expected:{expected})"
               )
            if result==expected:
                passed+=1
                print("pass")
            else:
                print("fail")

        print("-"*40)
        print(f"passed functions:{passed}/{len(test_cases)}")

        if passed==len(test_cases):
            print("overall result:correct")
        else:
            print("overall result:incorrect")

    except Exception as e:
        print("Error:", e)

print("generating test functions")
print("-"*40)

generated_code=generate_function()

print("\ngenerated the code")
print("-"*40)
print(generated_code)
print("-"*40)

test_function(generated_code)
