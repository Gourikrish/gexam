import ollama
import re

def generate_function():
    

    prompt ="""
    write a python function named is_prime(num).
    take the integer as arguments and return true 
    whether the number is prime, otheriwise return false.

    return only the prime function.
    do not give any explanations or markdowns.
    """
    response=ollama.generate(
        model="qwen2.5-coder:0.5b",
        prompt=prompt
    )
    generated_code=response["response"].strip()

    match=re.search(
        r"```(?:python)?\s*(.*?)```",
        generated_code,
        re.DOTALL
    )
    if match:
        generated_code=match.group(1).strip()
    return generated_code

def test_function(code):
    namespace={}

    try:
        exec(code, namespace)

        is_prime = namespace["is_prime"]

        test_cases=[
            (2, True),
            (3, True),
            (4, False),
            (5, True),
            (6, False)
        ]

        print("\ntesting generated function")
        print("-"*40)
        passed=0

        for num, expected in test_cases:
            result=is_prime(num)

            print(f"{num}={result}(Expected:{expected})")
            if result==expected:
                passed+=1
                print("pass")
            else:
                print("fail")

        print("-"*40)
        print(f"test cases passed:{passed}/{len(test_cases)}")

        if passed==len(test_cases):
            print("overall correct")
        else:
            print("overall fail")

    except Exception as e:
        print("error:",e)


print("ollama python function code generation")
print("-"*40)

generated_code=generate_function()

print(generated_code)
print("-"*40)

test_function(generated_code)
